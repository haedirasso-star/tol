🛡 Firestore Rules المحدّثة للنظام الجديد
Rules الرسمية لنظام activation_codes.
الضمانات: المستخدم لا يمنح نفسه بريميوم إلا بكود يملكه فعلاً · المستخدم لا يستطيع مسح اشتراكه أبداً (يمنع ضياع الاشتراك عند تسجيل الخروج) · الأكواد لا يمكن سردها أو تخمينها (list للأدمن فقط).
انسخها والصقها في Firebase Console ← Firestore ← Rules ← Publish. 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ═══════════════════════════════════════════════════════════════
    //  TOTV+ — قواعد الأمان  (نظام أكواد التفعيل)
    //  ملاحظات مهمة:
    //   • المستخدم لا يستطيع منح نفسه premium إلا بكود تفعيل يملكه فعلاً.
    //   • المستخدم لا يستطيع مسح اشتراكه أبداً (يمنع فقدان الاشتراك
    //     عند تسجيل الخروج — الأدمن وحده يستطيع الإلغاء).
    //   • أكواد التفعيل لا يمكن سردها (list) — فقط قراءة كود بمعرّفه.
    // ═══════════════════════════════════════════════════════════════

    // ─── Helpers ─────────────────────────────────────────────
    function isAuth()  { return request.auth != null; }
    function isOwner(uid) { return isAuth() && request.auth.uid == uid; }

    function isAdmin() {
      return isAuth() && (
        request.auth.token.email in ['admin@totv.com', 'haedirasso@gmail.com']
        || exists(/databases/$(database)/documents/admins/$(request.auth.token.email))
      );
    }

    // المفاتيح التي تغيّرت في هذه العملية
    function changed() {
      return request.resource.data.diff(resource.data).affectedKeys();
    }

    // هل يملك المستخدم الكود المذكور داخل الاشتراك الجديد؟
    function ownsCode(sub) {
      return sub.code is string
        && sub.code.size() >= 8
        && exists(/databases/$(database)/documents/activation_codes/$(sub.code))
        && get(/databases/$(database)/documents/activation_codes/$(sub.code))
             .data.used_by_uid == request.auth.uid;
    }

    // ─── app_config — الإعدادات العامة ───────────────────────
    // المستندات العامة يقرؤها الجميع، أما secrets فللأدمن فقط
    // (توكن تلجرام / مفاتيح API يجب ألا تُقرأ من التطبيق).
    match /app_config/secrets {
      allow read, write: if isAdmin();
    }
    match /app_config/{doc} {
      allow read:  if true;
      allow write: if isAdmin();
    }

    // ─── activation_codes — أكواد التفعيل ────────────────────
    //  get   : أي مستخدم مسجّل يقرأ كوداً واحداً بمعرّفه الدقيق فقط
    //  list  : ممنوع تماماً — لا يمكن سرد/تخمين الأكواد
    //  update: يُسمح فقط بحجز الكود لنفسه، بلا تغيير بيانات السيرفر
    match /activation_codes/{code} {
      allow get:    if isAuth();
      allow list:   if isAdmin();
      allow create: if isAdmin();
      allow delete: if isAdmin();

      allow update: if isAdmin() || (
        isAuth()
        && resource.data.status != 'disabled'
        && resource.data.disabled != true
        && (resource.data.used != true
            || resource.data.used_by_uid == request.auth.uid)
        && request.resource.data.used == true
        && request.resource.data.used_by_uid == request.auth.uid
        && request.resource.data.host     == resource.data.host
        && request.resource.data.username == resource.data.username
        && request.resource.data.password == resource.data.password
        && request.resource.data.plan     == resource.data.plan
      );
    }

    // ─── users/{uid} — بيانات المستخدم ───────────────────────
    match /users/{uid} {
      allow read:   if isOwner(uid) || isAdmin();
      allow create: if isOwner(uid);
      allow delete: if isAdmin();

      allow update: if isAdmin() || (
        isOwner(uid)
        && !changed().hasAny(['status', 'role', 'is_admin'])
        && (
             !changed().hasAny(['subscription'])
             || (request.resource.data.subscription.plan == 'premium'
                 && ownsCode(request.resource.data.subscription))
           )
      );
    }

    // ─── activations — سجل عمليات التفعيل ────────────────────
    match /activations/{id} {
      allow create: if isAuth() && request.resource.data.uid == request.auth.uid;
      allow read:   if isAdmin();
      allow update, delete: if isAdmin();
    }

    // ─── orders — طلبات الشراء/الحوالات ──────────────────────
    match /orders/{id} {
      allow read:   if isAdmin() ||
        (isAuth() && resource.data.uid == request.auth.uid);
      allow create: if isAuth() &&
        request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // ─── subscription_requests — طلبات من داخل التطبيق ───────
    match /subscription_requests/{id} {
      allow read:   if isAdmin() ||
        (isAuth() && resource.data.uid == request.auth.uid);
      allow create: if isAuth() &&
        request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // ─── complaints — الشكاوى ────────────────────────────────
    match /complaints/{id} {
      allow read:   if isAdmin() ||
        (isAuth() && resource.data.uid == request.auth.uid);
      allow create: if isAuth();
      allow update, delete: if isAdmin();
    }

    // ─── servers / credit_pool — تحوي كلمات مرور — للأدمن فقط ──
    match /servers/{id} {
      allow read, write: if isAdmin();
    }
    match /credit_pool/{id} {
      allow read, write: if isAdmin();
    }

    // ─── admins — قائمة المشرفين ─────────────────────────────
    match /admins/{email} {
      allow read:  if isAuth();
      allow write: if isAdmin();
    }

    // ─── notifications — قراءة للجميع ────────────────────────
    match /notifications/{doc} {
      allow read:  if true;
      allow write: if isAdmin();
    }

    // ─── online_users ─────────────────────────────────────────
    match /online_users/{uid} {
      allow read:  if isAuth();
      allow write: if isOwner(uid);
    }

    // ─── app_logs ─────────────────────────────────────────────
    match /app_logs/{log} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    // ─── Catch-all — منع كل شيء آخر ──────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
